# GuidePC 2 — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construir o sistema de monitoramento GuidePC 2 — coleta de métricas de sistema a cada 5s (Node/Express/systeminformation/SQLite) com 4 dashboards + alertas provisionados no Grafana (Docker) e página resumo leve.

**Architecture:** Processo Node é o **único escritor** de `data/metrics.db` (better-sqlite3, WAL); tudo o mais só lê — API Express (read-only) e Grafana com o plugin `frser-sqlite-datasource` montando `./data` como `:ro` (o plugin abre com `_pragma=query_only(1)`). Coletor normaliza métricas em tabela longa `samples(ts, metric, value)` com índices `ts` e `(metric, ts)`.

**Tech Stack:** Node 22 (ESM), Express 5, better-sqlite3, systeminformation 5, vitest, supertest, ESLint 9; Grafana (`grafana/grafana:latest`) via docker-compose com plugin `frser-sqlite-datasource`.

**Spec:** `docs/superpowers/specs/2026-10-05-guidepc2-design.md` (ler antes de cada task — o plano argumenta a partir dela)

## Global Constraints

- **Escritor único:** todo `INSERT`/`DELETE` em `metrics.db` passa SOMENTE por `src/db/writer.js`. Nenhum outro arquivo usa `db.prepare` de escrita. Grafana monta `./data:/data:ro`.
- `ts` armazenado em **epoch seconds** (exigência do plugin do Grafana; painéis filtram com `ts >= $__from / 1000 AND ts < $__to / 1000`).
- Intervalo de coleta: **5000ms**; retenção: **7 dias** (purge diário).
- Portas: **3001** = Express/API, **3000** = Grafana. Credenciais Grafana: `admin`/`guidepc2`.
- IDs fixos: datasource uid `guidepc2-metrics`; dashboards uid `guidepc2-cpu|memory|disk|network`.
- UI/textos em **PT-BR**; código/comentários/nomes em inglês.
- Testes com vitest; lint com ESLint flat config; commits convencionais (`feat:`, `test:`, `chore:`, `docs:`) após cada task.
- Métricas indisponíveis no host são **omitidas** (nunca gravadas como 0/NaN); coleta falha em um ciclo não derruba o scheduler.
- Sem dependências além das listadas (YAGNI).

```js
import fs from 'node:fs';
import path from 'node:path';
import Database from 'better-sqlite3';

const SCHEMA = `
CREATE TABLE IF NOT EXISTS samples (
  id     INTEGER PRIMARY KEY AUTOINCREMENT,
  ts     INTEGER NOT NULL,
  metric TEXT    NOT NULL,
  value  REAL    NOT NULL
);
CREATE INDEX IF NOT EXISTS idx_samples_ts ON samples(ts);
CREATE INDEX IF NOT EXISTS idx_samples_metric_ts ON samples(metric, ts);
`;

export function openDatabase(dbPath) {
  fs.mkdirSync(path.dirname(dbPath), { recursive: true });
  const db = new Database(dbPath);
  db.pragma('journal_mode = WAL');
  db.pragma('busy_timeout = 5000');
  db.pragma('synchronous = NORMAL');
  db.pragma('auto_vacuum = INCREMENTAL');
  db.exec(SCHEMA);
  return db;
}
```

- [ ] **Step 4: Implementar `src/db/writer.js` (único escritor)**

```js
export function createWriter(db) {
  const insert = db.prepare('INSERT INTO samples (ts, metric, value) VALUES (?, ?, ?)');
  const purge = db.prepare('DELETE FROM samples WHERE ts < ?');

  return {
    insertBatch(rows) {
      const tx = db.transaction((batch) => {
        for (const row of batch) insert.run(row.ts, row.metric, row.value);
      });
      tx(rows);
    },
    purgeOlderThan(cutoffTs) {
      const info = purge.run(cutoffTs);
      if (info.changes > 0) db.pragma('incremental_vacuum');
      return info.changes;
    },
  };
}
```

```js
export function createQueries(db) {
  const latestStmt = db.prepare(
    'SELECT metric, value, MAX(ts) AS ts FROM samples WHERE ts >= ? GROUP BY metric'
  );
  const rangeStmt = db.prepare(
    'SELECT ts, value FROM samples WHERE metric = ? AND ts >= ? AND ts < ? ORDER BY ts ASC'
  );
  const maxTsStmt = db.prepare('SELECT MAX(ts) AS t FROM samples');
  const countStmt = db.prepare(
    'SELECT COUNT(DISTINCT metric) AS c FROM samples WHERE ts >= ?'
  );

  return {
    latest(nowSec, windowSec = 120) {
      const rows = latestStmt.all(nowSec - windowSec);
      const values = {};
      let ts = null;
      for (const row of rows) {
        values[row.metric] = row.value;
        if (ts === null || row.ts > ts) ts = row.ts;
      }
      return { ts, values };
    },
    range(metric, fromSec, toSec) {
      return { metric, points: rangeStmt.all(metric, fromSec, toSec) };
    },
    maxTs() {
      return maxTsStmt.get().t ?? null;
    },
    countRecent(windowSec, nowSec = Math.floor(Date.now() / 1000)) {
      return countStmt.get(nowSec - windowSec).c;
    },
  };
}
```

```js
import os from 'node:os';

const round = (n, digits = 1) => Number(n.toFixed(digits));
const toGb = (bytes) => round(bytes / 1024 ** 3, 2);
const toMbPerSec = (bytesPerSec) => round(bytesPerSec / 1024 ** 2, 2);

export async function collectMetrics(si, state, nowMs = Date.now()) {
  const out = [];
  const push = (metric, value) => {
    if (typeof value !== 'number' || !Number.isFinite(value)) return;
    out.push({ metric, value });
  };

  try {
    const load = await si.currentLoad();
    push('cpu.total', round(load.currentLoad));
    (load.cpus ?? []).forEach((cpu, i) => push(`cpu.core.${i}`, round(cpu.load)));
  } catch {
    /* omitido */
  }

  if (process.platform !== 'win32') {
    const [l1, l5, l15] = os.loadavg();
    push('cpu.load.1', round(l1, 2));
    push('cpu.load.5', round(l5, 2));
    push('cpu.load.15', round(l15, 2));
  }

  try {
    const temp = await si.cpuTemperature();
    if (temp && temp.main > 0) push('cpu.temp', round(temp.main));
  } catch {
    /* omitido */
  }

  try {
    const mem = await si.mem();
    if (mem.total > 0) push('mem.used', round((mem.active / mem.total) * 100));
    push('mem.available', toGb(mem.available));
    if (mem.swaptotal > 0) push('mem.swap', round((mem.swapused / mem.swaptotal) * 100));
    push('mem.cached', toGb(mem.buffcache ?? 0));
  } catch {
    /* omitido */
  }

  try {
    const filesystems = await si.fsSize();
    for (const fs of filesystems) {
      const key = String(fs.mount || fs.fs).replace(/[\\/\s]/g, '_');
      if (typeof fs.use === 'number') push(`disk.usage.${key}`, round(fs.use));
      if (typeof fs.available === 'number') push(`disk.free.${key}`, toGb(fs.available));
    }
  } catch {
    /* omitido */
  }

  if (process.platform !== 'win32') {
    try {
      const io = await si.fsStats();
      if (typeof io.rx_sec === 'number') push('disk.read', toMbPerSec(io.rx_sec));
      if (typeof io.wx_sec === 'number') push('disk.write', toMbPerSec(io.wx_sec));
    } catch {
      /* omitido */
    }
  }

  try {
    const stats = await si.networkStats('*');
    const prev = state.net ?? {};
    const current = {};
    let totalErrors = 0;
    for (const s of stats) {
      current[s.iface] = { rx: s.rx_bytes, tx: s.tx_bytes, at: nowMs };
      const before = prev[s.iface];
      if (before && nowMs > before.at) {
        const dtSec = (nowMs - before.at) / 1000;
        const rxDelta = s.rx_bytes - before.rx;
        const txDelta = s.tx_bytes - before.tx;
        push(`net.rx.${s.iface}`, round((rxDelta > 0 ? rxDelta : 0) / dtSec / 1024, 2));
        push(`net.tx.${s.iface}`, round((txDelta > 0 ? txDelta : 0) / dtSec / 1024, 2));
      }
      totalErrors +=
        (s.rx_errors ?? 0) + (s.tx_errors ?? 0) + (s.rx_dropped ?? 0) + (s.tx_dropped ?? 0);
    }
    if (stats.length > 0) push('net.errors', totalErrors);
    state.net = current;
  } catch {
    /* omitido */
  }

  try {
    const connections = await si.networkConnections();
    push(
      'net.conns.established',
      connections.filter((c) => c.state === 'ESTABLISHED').length
    );
  } catch {
    /* omitido */
  }

  return out;
}
```

Nota: `net.errors` é o **total acumulado desde o boot** (soma de todos os contadores de erro/drop de todas as interfaces) — é uma única métrica global, calculada depois do loop para não gerar entradas duplicadas.

```js
import { collectMetrics } from './metrics.js';

export function createCollector({ si, writer, intervalMs, onError = console.error }) {
  const state = {};
  let timer = null;
  let running = false;

  async function runOnce(nowMs = Date.now()) {
    const samples = await collectMetrics(si, state, nowMs);
    if (samples.length > 0) {
      const ts = Math.floor(nowMs / 1000);
      writer.insertBatch(samples.map((s) => ({ ts, metric: s.metric, value: s.value })));
    }
    return samples;
  }

  async function tick() {
    if (running) return;
    running = true;
    try {
      await runOnce();
    } catch (error) {
      onError(error);
    } finally {
      running = false;
    }
  }

  return {
    start() {
      if (timer === null) timer = setInterval(tick, intervalMs);
    },
    stop() {
      if (timer !== null) {
        clearInterval(timer);
        timer = null;
      }
    },
    runOnce,
  };
}
```

```js
import { Router } from 'express';

export function createApiRouter({ queries }) {
  const router = Router();

  router.get('/health', (_req, res) => {
    res.json({
      status: 'ok',
      uptime: Math.round(process.uptime()),
      lastCollectionAt: queries.maxTs(),
      metricsCount: queries.countRecent(120),
    });
  });

  router.get('/latest', (_req, res) => {
    res.json(queries.latest(Math.floor(Date.now() / 1000)));
  });

  router.get('/metrics/:metric', (req, res) => {
    const now = Math.floor(Date.now() / 1000);
    const from = req.query.from !== undefined ? Number(req.query.from) : now - 3600;
    const to = req.query.to !== undefined ? Number(req.query.to) : now + 1;
    if (!Number.isFinite(from) || !Number.isFinite(to) || from >= to) {
      return res.status(400).json({ error: 'Parâmetros from/to inválidos' });
    }
    return res.json(queries.range(req.params.metric, from, to));
  });

  router.post('/alerts/webhook', (req, res) => {
    console.log('[alerta Grafana]', JSON.stringify(req.body));
    res.status(200).json({ ok: true });
  });

  return router;
}
```
